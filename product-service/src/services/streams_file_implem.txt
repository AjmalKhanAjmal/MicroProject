# 1️⃣ Upload Excel via **Multer**

### middleware

```js
// middlewares/upload.middleware.js
const multer = require("multer");
const path = require("path");

const storage = multer.diskStorage({
  destination: "uploads/",
  filename: (_, file, cb) => {
    cb(null, `${Date.now()}-${file.originalname}`);
  },
});

module.exports = multer({
  storage,
  fileFilter: (_, file, cb) => {
    if (!file.originalname.endsWith(".xlsx")) {
      return cb(new Error("Only Excel files allowed"));
    }
    cb(null, true);
  },
});
```

# 5️⃣ Excel Reader (Duplicate Check by SKU)

```js
// utils/excelImporter.js
const ExcelJS = require("exceljs");

async function readExcel(filePath, onRow) {
  const workbook = new ExcelJS.stream.xlsx.WorkbookReader(filePath);

  for await (const sheet of workbook) {
    let header = true;

    for await (const row of sheet) {
      if (header) {
        header = false;
        continue;
      }

      await onRow({
        sku: row.getCell(1).value,
        name: row.getCell(2).value,
        price: row.getCell(3).value,
        stock: row.getCell(4).value,
      });
    }
  }
}

module.exports = { readExcel };
```

# 6️⃣ Background Job Processor (🔥 MAIN LOGIC)

```js
// jobs/productImport.job.js
const { readExcel } = require("../utils/excelImporter");
const logger = require("../utils/logger");
const { Product, ImportStatus, sequelize } = require("../models");

module.exports = async (job) => {
  const { filePath, importId } = job.data;
  const transaction = await sequelize.transaction();

  try {
    await ImportStatus.update(
      { status: "PROCESSING" },
      { where: { id: importId } }
    );

    let processed = 0;

    await readExcel(filePath, async (row) => {
      const exists = await Product.findOne({
        where: { sku: row.sku },
        transaction,
      });

      if (!exists) {
        await Product.create(row, { transaction });
      }

      processed++;
      await ImportStatus.update(
        { processedRows: processed },
        { where: { id: importId } }
      );
    });

    await transaction.commit();

    await ImportStatus.update(
      { status: "COMPLETED" },
      { where: { id: importId } }
    );

    logger.info("Import completed", { importId });
  } catch (err) {
    await transaction.rollback();

    await ImportStatus.update(
      { status: "FAILED", error: err.message },
      { where: { id: importId } }
    );

    logger.error("Import failed", err);
  }
};
```

# 7️⃣ Controller (API stays FAST 🚀)

```js
// controllers/productImport.controller.js
const productQueue = require("../queues/product.queue");
const { ImportStatus } = require("../models");

exports.uploadExcel = async (req, res) => {
  const record = await ImportStatus.create({
    fileName: req.file.filename,
    processedRows: 0,
    status: "PENDING",
  });

  await productQueue.add({
    filePath: req.file.path,
    importId: record.id,
  });

  res.json({
    message: "Import started in background",
    importId: record.id,
  });
};
```
