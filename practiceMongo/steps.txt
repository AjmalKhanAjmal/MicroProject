const createUser = async () => {
  const response = await fetch(
    "http://localhost:9000/api/users",
    {
      method: "POST",
      headers,
      body: JSON.stringify({
        name: "Ajmal",
        email: "ajmal@gmail.com",
        role: "user"
      })
    }
  );

  if (!response.ok) throw new Error("Create failed");

  const data = await response.json();

};
