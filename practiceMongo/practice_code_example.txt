function proceedToPayment(cart_id, callback) {

    (!cart_id) ? callback("cart_id required") : ""

    setTimeout(() => {
        callback(null, {
            "data": {
                message: "payment done",
                cart_id: cart_id
            }
        })
    }, 1000)

}

// Simulating a real checkout for cart_id "1234"
function checkout(cart_id) {
    console.log("Starting checkout for cart:", cart_id);

    proceedToPayment(cart_id, (paymentError, paymentData) => {
        if (paymentError) {
            return console.error("Payment Error:", paymentError);
        }
        console.log("Payment successful:", paymentData.data.cart_id);

        createOrder(paymentData.data.cart_id, (orderError, orderData) => {
            if (orderError) {
                return console.error("Order Creation Error:", orderError);
            }
            console.log("Order created with ID:", orderData.data.order_id);

            sendEmail(orderData, (emailError, emailStatus) => {
                if (emailError) {
                    return console.error("Email Error:", emailError);
                }
                console.log("Checkout Complete! Email Status:", emailStatus);
            });
        });
    });
}




// Run the checkout
checkout("1234");
